cmake_minimum_required(VERSION 3.10)
project(es_audit_regressions LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(AUDIT_SANITIZERS "Enable address and undefined behavior sanitizers" ON)
find_package(CURL REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL2 REQUIRED sdl2)
set(CORE "${CMAKE_CURRENT_LIST_DIR}/../../es-core/src")
add_executable(audit_regressions regressions.cpp support.cpp
  ${CORE}/HttpReq.cpp ${CORE}/utils/FileSystemUtil.cpp
  ${CORE}/utils/StringUtil.cpp ${CORE}/utils/ZipFile.cpp ${CORE}/utils/md5.cpp)
target_include_directories(audit_regressions PRIVATE ${CORE} ${SDL2_INCLUDE_DIRS})
target_compile_options(audit_regressions PRIVATE -ffunction-sections -fdata-sections)
target_link_libraries(audit_regressions PRIVATE CURL::libcurl pthread)
set_property(TARGET audit_regressions APPEND_STRING PROPERTY LINK_FLAGS " -Wl,--gc-sections")
if(AUDIT_SANITIZERS)
  target_compile_options(audit_regressions PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_libraries(audit_regressions PRIVATE -fsanitize=address,undefined)
endif()
